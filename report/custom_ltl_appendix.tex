\section{Custom Question Bank LTL Specifications}
\label{app:custom_ltl}

This appendix lists the three LTL properties for each custom question. Atoms, assumptions and machine-readable formulas are stored in \texttt{Cxx.ltl.json}. The operators are: $\mathbf{G}$ (always), $\mathbf{F}$ (eventually), $\mathbf{X}$ (next scan), $\wedge$ (and), $\vee$ (or), $\neg$ (not) and $\rightarrow$ (implies).

\begin{table}[htbp]
\centering
\small
\caption{Derived and internal atoms used in the LTL properties.}
\label{tab:custom_atoms}
\begin{tabular}{|l|l|p{0.58\textwidth}|}
\hline
Atom & Kind & Definition \\
\hline
\texttt{LeftOnlyTimerQ} & internal & \texttt{leftTimer.Q}: left button held alone for 500\,ms \\
\texttt{RightOnlyTimerQ} & internal & \texttt{rightTimer.Q}: right button held alone for 500\,ms \\
\texttt{HeightInRange} & derived & $40.0 \le \mathrm{HeightSensor} \le 60.0$ \\
\texttt{HeightOutOfRange} & derived & Height outside $[40.0, 60.0]$ but inside the valid sensor range $[0.0, 100.0]$ \\
\texttt{LayerComplete} & internal & \texttt{layerTrigger}: third box of a layer has been detected \\
\hline
\end{tabular}
\end{table}

All other atoms are interface variables defined in the corresponding question.

\subsection*{C01 --- Product Reject and Filtering Station}
\noindent POU: \texttt{FB\_C01}; file: \texttt{C01.ltl.json}.
\begin{small}
\begin{itemize}
\item \textbf{P1 --- Defective part is rejected}: $\mathbf{G}((PartPresent \wedge DefectSensor \wedge RunSwitch \wedge \neg BatchFull) \rightarrow RejectPusher)$
\item \textbf{P2 --- Pusher retracts}: $\mathbf{G}(RejectPusher \rightarrow \mathbf{F} \neg RejectPusher)$
\item \textbf{P3 --- Batch full stops the conveyor}: $\mathbf{G}(BatchFull \rightarrow \neg ConveyorMotor)$
\end{itemize}
\end{small}

\subsection*{C02 --- Two-Hand Safety Press}
\noindent POU: \texttt{FB\_C02}; file: \texttt{C02.ltl.json}.
